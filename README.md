#include <stdio.h>

int main() {
    int num1, num2, soma;

    // Lendo dois numeros de usuario 
    printf("Digite o primeiro numero: );
    scanf("%d", &num1);

    printf("digite o segundo numero: );
    scanf("%d", &num2);

    // Calculando a soma de dois numeros
    soma = num1 + num2;

    // Exibindo o resultado 
    Printf("A soma de %d e %d é: %d\n", num1, num2, soma);

    return 0;
    
}
